\begin{frame}{Project Overview}
\small
\begin{columns}[T,onlytextwidth]
  \column{0.52\textwidth}
  \begin{block}{What is our project?}
  We study how volatility shocks propagate across six major equity markets using a Bayesian VAR, and we compare calm and crisis regimes to see whether global transmission intensifies under stress.
  \end{block}

  \begin{block}{Empirical design}
  \begin{itemize}
    \item Six implied-volatility indices in levels
    \item Full sample plus calm-vs-crisis comparison
    \item IRFs, FEVDs, and conditional forecasts as the main outputs
  \end{itemize}
  \end{block}

  \column{0.44\textwidth}
  \begin{block}{Why this is a good \texttt{BVAR} application}
  \begin{itemize}
    \item Volatility is persistent and strongly comoving.
    \item A six-variable VAR with five lags already has 186 coefficients.
    \item Bayesian shrinkage is therefore central to making the model usable.
  \end{itemize}
  \end{block}

  \begin{alertblock}{Our angle}
  We use this project to show how the \texttt{BVAR} package turns a theoretically rich model into an applied workflow that is easy to estimate, diagnose, and interpret.
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Why \texttt{BVAR}?}
\small
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \begin{block}{Dimensionality problem}
  A VAR with $M$ variables and $p$ lags has
  \[
    M + M^2 p
  \]
  slope parameters before the covariance matrix.

  With $M=6$ and $p=5$, this already implies
  \[
    6 + 6^2 \cdot 5 = 186
  \]
  coefficients to regularise.
  \end{block}

  \column{0.48\textwidth}
  \begin{block}{Why Bayesian shrinkage helps}
  \begin{itemize}
    \item It shrinks weak coefficients toward sensible benchmarks.
    \item It reduces overfitting in a medium-size system.
    \item It improves stability for forecasting and impulse analysis.
  \end{itemize}
  \end{block}
\end{columns}

\vspace{0.4em}
\begin{alertblock}{Practical message}
For this kind of volatility system, Bayesian regularisation is not just elegant theory; it is what makes the VAR empirically tractable.
\end{alertblock}
\end{frame}

\begin{frame}{Compact Theory: VAR and Minnesota Prior}
\scriptsize
\begin{columns}[T,onlytextwidth]
  \column{0.52\textwidth}
  \begin{block}{Bayesian VAR}
  \[
    \boldsymbol{y}_{t}=\boldsymbol{a}_0+A_1\boldsymbol{y}_{t-1}+\cdots+A_p\boldsymbol{y}_{t-p}+\boldsymbol{\varepsilon}_t,
    \qquad
    \boldsymbol{\varepsilon}_t\sim\mathcal{N}(\boldsymbol{0},\Sigma)
  \]
  \end{block}

  \begin{block}{Minnesota prior}
  \[
    \mathbb{E}\!\left[(A_s)_{ij}\mid\Sigma\right]=
    \begin{cases}
      1 & i=j,\ s=1 \\
      0 & \text{otherwise}
    \end{cases}
  \]
  \[
    \operatorname{Var}\!\left[(A_s)_{ij}\mid\Sigma\right]
    \propto
    \frac{\lambda^2\psi_j}{s^{2\alpha}}
    \cdot
    \frac{\sigma_{ii}}{\sigma_{jj}}
  \]
  \end{block}

  \column{0.44\textwidth}
  \begin{block}{Interpretation}
  \begin{itemize}
    \item Each series is assumed to be persistent in its own first lag.
    \item Cross-variable lag effects are shrunk toward zero unless the data strongly support them.
    \item $\lambda$ controls overall tightness, $\alpha$ controls lag decay, and $\psi_j$ scales variable-specific uncertainty.
  \end{itemize}
  \end{block}

  \begin{alertblock}{Why this fits our data}
  Volatility indices are persistent and correlated, so the Minnesota prior stabilises the system without ruling out spillovers.
  \end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Compact Theory: Learning Shrinkage from the Data}
\small
\begin{block}{Hierarchical prior selection}
\[
  p(\boldsymbol{\theta},\boldsymbol{\gamma}\mid\boldsymbol{y})
  \propto
  p(\boldsymbol{y}\mid\boldsymbol{\gamma})\,p(\boldsymbol{\gamma}),
  \qquad
  p(\boldsymbol{y}\mid\boldsymbol{\gamma})
  =
  \int
  p(\boldsymbol{y}\mid\boldsymbol{\theta},\boldsymbol{\gamma})
  p(\boldsymbol{\theta}\mid\boldsymbol{\gamma})
  \,d\boldsymbol{\theta}
\]
\end{block}

\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \begin{block}{Interpretation}
  \begin{itemize}
    \item Hyperparameters such as $\lambda$ are not fixed manually.
    \item They are learned through the marginal likelihood.
    \item The procedure automatically balances fit against overfitting.
  \end{itemize}
  \end{block}

  \column{0.48\textwidth}
  \begin{block}{Why \texttt{BVAR} helps}
  \begin{itemize}
    \item The package estimates the model, samples hyperparameters, and produces diagnostics in one workflow.
    \item IRFs, FEVDs, and scenario forecasts are all generated from the fitted object.
    \item This makes the methodology much easier to operationalise in practice.
  \end{itemize}
  \end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile]{\texttt{BVAR} Package Workflow}

mn <- bv_minnesota(
  lambda = bv_lambda(mode = 0.2, sd = 0.4),
  alpha  = bv_alpha(mode = 2),
  psi    = bv_psi(scale = 0.004, shape = 0.004)
)

fit <- bvar(
  data = data_full, lags = 5,
  n_draw = 25000, n_burn = 10000,
  priors = bv_priors(hyper = c("lambda", "alpha", "psi"), mn = mn),
  irf = bv_irf(horizon = 20, fevd = TRUE, identification = TRUE),
  fcast = bv_fcast(horizon = 20)
)

irf_full  <- irf(fit, horizon = 20, conf_bands = c(0.05, 0.16))
fevd_full <- fevd(irf_full, conf_bands = c(0.16))
pred_full <- predict(fit, conf_bands = c(0.05, 0.16))
\end{lstlisting}

\begin{frame}{SOC and SUR: What Do They Add?}
\small
\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \begin{block}{Sum-of-coefficients prior (SOC)}
  \[
    \boldsymbol{y^+}=diag\!\left(\frac{\bar{\boldsymbol{y}}}{\mu}\right),
    \qquad
    \boldsymbol{x^+}=[\boldsymbol{0},\boldsymbol{y^+},\dots,\boldsymbol{y^+}]
  \]
  \end{block}

  \column{0.48\textwidth}
  \begin{block}{Single-unit-root prior (SUR)}
  \[
    \boldsymbol{y^{++}}=\frac{\bar{\boldsymbol{y}}^\top}{\delta},
    \qquad
    \boldsymbol{x^{++}}=\left[\frac{1}{\delta},\boldsymbol{y^{++}},\dots,\boldsymbol{y^{++}}\right]
  \]
  \end{block}
\end{columns}

\vspace{0.4em}
\begin{block}{Interpretation}
\begin{itemize}
  \item SOC pushes the system toward persistent no-change dynamics.
  \item SUR allows common stochastic trends or cointegration-like persistence.
  \item Because our series are volatility levels, these priors are admissible as a robustness extension.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Project Conclusions}
\small
\begin{itemize}
  \item A US volatility shock propagates positively to all markets in the system, so spillovers are clearly global.
  \item The transmission is stronger and more uncertain during the 2020 crisis window than in the calm period.
  \item VIX is the dominant driver in both IRFs and FEVDs, while India remains comparatively more self-driven and less system-defining.
  \item Conditional forecasting confirms this asymmetry: a VIX stress scenario moves the whole system more than an India stress scenario.
  \item Our main findings are robust to lag choice and remain qualitatively unchanged when SOC and SUR are added.

\begin{frame}{Appendix: VAR Setup}
\small
\begin{block}{Model}
\[
  \boldsymbol{y}_{t} = \boldsymbol{a}_0 + A_1 \boldsymbol{y}_{t-1} + \cdots + A_p \boldsymbol{y}_{t-p} + \boldsymbol{\varepsilon}_t,
  \qquad
  \boldsymbol{\varepsilon}_t \sim \mathcal{N}(\boldsymbol{0}, \Sigma)
\]
\end{block}

\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \begin{block}{Why OLS is not enough}
  \begin{itemize}
    \item Parameter count rises quickly with the number of variables and lags.
    \item Financial volatility series are persistent and strongly correlated.
    \item Unregularised VARs can overfit in-sample noise.
  \end{itemize}
  \end{block}

  \column{0.48\textwidth}
  \begin{block}{Why Bayesian shrinkage helps}
  \begin{itemize}
    \item Priors shrink weak coefficients toward economically sensible benchmarks.
    \item The data can still dominate when the signal is strong.
    \item This improves stability and forecast performance in medium-size systems.
  \end{itemize}
  \end{block}
\end{columns}
\end{frame}

\begin{frame}{Appendix: Minnesota Prior}
\small
\begin{block}{Prior mean}
\[
  \mathbb{E}\!\left[(A_s)_{ij}\mid\Sigma\right] =
  \begin{cases}
    1 & \text{if } i=j,\ s=1 \\
    0 & \text{otherwise}
  \end{cases}
\]
\end{block}

\begin{block}{Prior variance}
\[
  \operatorname{Var}\!\left[(A_s)_{ij}\mid\Sigma\right]
  \propto
  \frac{\lambda^2 \psi_j}{s^{2\alpha}}
  \cdot
  \frac{\sigma_{ii}}{\sigma_{jj}}
\]
\end{block}

\begin{itemize}
  \item $\lambda$: overall tightness
  \item $\alpha$: lag decay
  \item $\psi_j$: variable-specific scaling
  \item Intuition: own first lags are persistent; cross-lag effects are shrunk strongly toward zero
\end{itemize}
\end{frame}

\begin{frame}{Appendix: Hierarchical Hyperparameter Selection}
\small
\begin{block}{Key idea}
Treat the prior hyperparameters as random variables:
\[
  p(\boldsymbol{\theta}, \boldsymbol{\gamma} \mid \boldsymbol{y})
  \propto
  p(\boldsymbol{y} \mid \boldsymbol{\gamma})\,
  p(\boldsymbol{\gamma}),
  \qquad
  p(\boldsymbol{y} \mid \boldsymbol{\gamma})
  =
  \int
  p(\boldsymbol{y} \mid \boldsymbol{\theta}, \boldsymbol{\gamma})
  p(\boldsymbol{\theta} \mid \boldsymbol{\gamma})
  \, d\boldsymbol{\theta}
\]
\end{block}

\begin{columns}[T,onlytextwidth]
  \column{0.48\textwidth}
  \begin{block}{Why this matters}
  \begin{itemize}
    \item Shrinkage is learned from the data instead of fixed by hand.
    \item The marginal likelihood penalises overly flexible models automatically.
    \item Results become easier to justify and replicate.
  \end{itemize}
  \end{block}

  \column{0.48\textwidth}
  \begin{block}{How \texttt{BVAR} implements it}
  \begin{itemize}
    \item Conjugate NIW structure gives closed-form posteriors for core parameters.
    \item Hyperparameters are sampled with Metropolis-Hastings.
    \item Diagnostics are available directly from the fitted object.
  \end{itemize}
  \end{block}
\end{columns}
\end{frame}

\begin{frame}{Appendix: SOC and SUR Dummy Priors}
\small
\begin{block}{Sum-of-coefficients prior (SOC)}
\[
  \boldsymbol{y^+} = diag\!\left(\frac{\bar{\boldsymbol{y}}}{\mu}\right),
  \qquad
  \boldsymbol{x^+} = [\boldsymbol{0}, \boldsymbol{y^+}, \dots, \boldsymbol{y^+}]
\]
\end{block}

\begin{block}{Single-unit-root prior (SUR)}
\[
  \boldsymbol{y^{++}} = \frac{\bar{\boldsymbol{y}}^\top}{\delta},
  \qquad
  \boldsymbol{x^{++}} = \left[\frac{1}{\delta}, \boldsymbol{y^{++}}, \dots, \boldsymbol{y^{++}}\right]
\]
\end{block}

\begin{itemize}
  \item SOC shrinks the system toward ``no large changes from current levels.''
  \item SUR allows persistent common movement and is useful when cointegration-like behaviour is plausible.
  \item These priors are designed for VARs in levels, which is why we only use them as a level-data robustness extension.
\end{itemize}
\end{frame}
